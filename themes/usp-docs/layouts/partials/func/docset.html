{{- /* The version directory a page belongs to, or false for pages outside one. A site with several products keeps its versions below them, as /waap/latest; a single-product site keeps them at the root, as /latest. */ -}} {{- $depth := 2 -}} {{- with site.GetPage "/latest" }}{{ $depth = 1 }}{{ end -}} {{- $docset := false -}} {{- $parts := split (strings.Trim .Path "/") "/" -}} {{- if and (index $parts 0) (ge (len $parts) $depth) -}} {{- with site.GetPage (printf "/%s" (delimit (first $depth $parts) "/")) -}} {{- /* A docset is a directory; the error page sits at that depth too. */ -}} {{- if eq .Kind "section" }}{{ $docset = . }}{{ end -}} {{- end -}} {{- end -}} {{- return $docset -}}
